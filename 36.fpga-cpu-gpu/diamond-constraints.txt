
# Restricciones solo para el flujo de Diamond (tools/diamond_runner.py las anexa al .lpf).
#
# TRACE trata como relacionados los relojes que salen de PLL distintos (clk de pll_cpu,
# clk_mem de pll_mem, clk_pix/clk_pix_5x de clock2_gen) y del pad clk_25mhz (GPU), y suma
# desfases de PLL como skew: da errores en cruces que no existen en el chip. Todos los cruces
# entre dominios son asincronos por diseno:
#
# 1) clk <-> clk_25mhz_c (CPU <-> GPU): gpu_mmio_bridge.v. Toggle de peticion/confirmacion con
#    2 FF en cada sentido; address/write_data estables desde antes del toggle (dato con toggle).
#
# 2) clk <-> clk_mem y clk_25mhz_c <-> clk_mem (maestros <-> SDRAM): fabric_fifo_bridge.v, que
#    usa async_fifo.v (punteros Gray con 2 FF).
#
# 3) clk <-> clk_pix y clk_pix -> clk_pix_5x: video_scanout.v (toggles por 2 FF, banco con
#    toggle) y consola; el mismo razonamiento que 30.fpga-cpu-console/diamond-constraints.txt.
#
# Lo que NO se bloquea es clk -> clk, clk_mem -> clk_mem ni clk_25mhz_c -> clk_25mhz_c: son los
# caminos que de verdad limitan la frecuencia.
BLOCK PATH FROM CLKNET "clk" TO CLKNET "clk_25mhz_c" ;
BLOCK PATH FROM CLKNET "clk_25mhz_c" TO CLKNET "clk" ;
BLOCK PATH FROM CLKNET "clk" TO CLKNET "clk_mem" ;
BLOCK PATH FROM CLKNET "clk_mem" TO CLKNET "clk" ;
BLOCK PATH FROM CLKNET "clk_25mhz_c" TO CLKNET "clk_mem" ;
BLOCK PATH FROM CLKNET "clk_mem" TO CLKNET "clk_25mhz_c" ;
BLOCK PATH FROM CLKNET "clk" TO CLKNET "clk_pix" ;
BLOCK PATH FROM CLKNET "clk_pix" TO CLKNET "clk" ;
BLOCK PATH FROM CLKNET "clk_pix" TO CLKNET "clk_pix_5x" ;
